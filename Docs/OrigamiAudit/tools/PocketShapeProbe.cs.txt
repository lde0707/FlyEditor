using System;using System.IO;using System.Reflection;using UnityEngine;using FlyEditor.Flight;
public static class PocketShapeProbe {
 static Vector2 V(float x,float y)=>new Vector2(x,y);
 static Vector2 Local(PaperSheet.Face f,Vector2 p){var v=f.material.inverse.MultiplyPoint3x4(new Vector3(p.x,0,p.y));return new Vector2(v.x,v.z);}
 public static bool Fold(PaperSheet sheet,string name,Vector2 tip,Vector2 stop,Vector2 l,Vector2 r,int sign){
  for(int i=0;i<sheet.Faces.Count;i++){
   var f=sheet.Faces[i];if(!f.Contains(Local(f,tip)))continue;
   var step=new FoldStep{cornerPull=true,customFlanks=true,flexible=true,closeFlat=true,regionalPull=true,coupledHinges=true,anchor=i,flap=i+1,pullTip=Local(f,tip),pullStop=Local(f,stop),pullLeft=Local(f,l),pullRight=Local(f,r)};
   var plan=PaperCornerPull.Create(sheet,step);if(plan==null)continue;
   var watch=System.Diagnostics.Stopwatch.StartNew();plan.TryMove(0,sign*180,out float reached);watch.Stop();
   Debug.Log("[ShapeJoint] "+name+" anchor="+i+" reached="+reached+" ms="+watch.ElapsedMilliseconds);
   if(Mathf.Abs(reached)<179.99f)continue;
   if(!sheet.FinishPull(plan,reached))continue;
   FlyEditor.Editor.FourPointFoldingVerification.Metrics(sheet);
   var replay=PaperSheet.Replay(JsonUtility.FromJson<ResearchSave>(JsonUtility.ToJson(new ResearchSave{paperFolds=sheet.Recipe()})).paperFolds);
   float error=0;if(replay.Faces.Count!=sheet.Faces.Count)throw new Exception("Replay topology");
   for(int j=0;j<sheet.Faces.Count;j++){var a=sheet.Faces[j];var b=replay.Faces[j];for(int k=0;k<a.points.Length;k++)error=Mathf.Max(error,Vector3.Distance(a.World(a.points[k],0),b.World(b.points[k],0)));}
   Debug.Log("[ShapeJoint] geometry PASS replay error="+error+" faces="+sheet.Faces.Count);Render(sheet,name);return true;
  }return false;
 }
 static void Render(PaperSheet sheet,string name){var flags=BindingFlags.Static|BindingFlags.NonPublic;var folder=Path.GetFullPath(Path.Combine(Application.dataPath,"../../../../.pocket-system/shapes"));Directory.CreateDirectory(folder);typeof(OrigamiFeasibilityAudit).GetField("folder",flags).SetValue(null,folder);typeof(OrigamiFeasibilityAudit).GetMethod("Render",flags).Invoke(null,new object[]{sheet,name,null});File.WriteAllText(Path.Combine(folder,name+".json"),JsonUtility.ToJson(new ResearchSave{paperFolds=sheet.Recipe()},true));}
 public static void Run(){
  var p=new PaperSheet();p.ApplyStep(new FoldStep(CreaseFamily.Falling,11,1));p.ApplyStep(new FoldStep(CreaseFamily.Rising,0,-1));
  if(Fold(p,"diagonal-front",V(0,11),V(5.5f,5.5f),V(0,5.5f),V(5.5f,11),1))Fold(p,"diagonal-back",V(11,0),V(5.5f,5.5f),V(5.5f,0),V(11,5.5f),-1);
  foreach(int sign in new[]{1,-1}){
   var q=new PaperSheet();q.ApplyStep(new FoldStep(CreaseFamily.Vertical,0,1){creaseNormal=Vector2.right,creaseOffset=5.5f});q.ApplyStep(new FoldStep(CreaseFamily.Horizontal,0,1){creaseNormal=Vector2.up,creaseOffset=5.5f});
   Fold(q,"axis-pocket-"+sign,V(5.5f,11),V(5.5f,5.5f),V(0,11),V(11,11),sign);
   var h=new PaperSheet();h.ApplyStep(new FoldStep(CreaseFamily.Vertical,0,1){creaseNormal=Vector2.right,creaseOffset=5.5f});
   Fold(h,"half-head-"+sign,V(5.5f,11),V(5.5f,8.25f),V(2.75f,11),V(8.25f,11),sign);
  }
 }
}
